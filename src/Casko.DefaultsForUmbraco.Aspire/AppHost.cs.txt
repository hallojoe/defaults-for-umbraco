var builder = DistributedApplication.CreateBuilder(args);
var sqlOnly = IsEnabled("SST_APPHOST_SQL_ONLY");
var install = IsEnabled("SST_APPHOST_INSTALL");

var sql = builder
    .AddSqlServer("sql")
    .WithImage("azure-sql-edge")
    .WithImageRegistry("mcr.microsoft.com")
    .WithHostPort(11433)
    .WithDataVolume("sst-sql-data");

if (sqlOnly)
{
    builder.Build().Run();
    return;
}

var distributedCacheDb = sql
    .AddDatabase("distributedCacheDbDSN", "sst-local-distributed-cache")
    .WithCreationScript(GetSqlDistributedCacheCreationScript());

if (install)
{
    builder.AddProject<Projects.Sst_AppHost_DatabaseImport>("import-umbraco-db")
        .WithEnvironment("ConnectionStrings__sql", sql)
        .WithEnvironment("DatabaseImport__DatabaseName", "sst-local-database")
        .WithEnvironment("DatabaseImport__BacpacPath", Path.GetFullPath(Path.Combine(builder.AppHostDirectory, "..", "sst-local-database.bacpac")))
        .WaitFor(distributedCacheDb);

    builder.Build().Run();
    return;
}

var umbracoDb = sql.AddDatabase("umbracoDbDSN", "sst-local-database");
var schedulingPublisherTempPath = Path.GetFullPath(Path.Combine(builder.AppHostDirectory, "..", "aspnet", "temp", "schedulingpublisher"));
var subscriberTempPath = Path.GetFullPath(Path.Combine(builder.AppHostDirectory, "..", "aspnet", "temp", "subscriber"));

var cms = builder.AddProject<Projects.Sst_Cms_Web_UI>("cms-web-ui", launchProfileName: "Umbraco.Web.UI.SchedulingPublisher")
    .WithEnvironment("ASPNETCORE_ENVIRONMENT", "Local")
    .WithEnvironment("DOTNET_ENVIRONMENT", "Local")
    .WithEnvironment("UMBRACO_SERVER_ROLE", "CM")
    .WithEnvironment("TEMP", schedulingPublisherTempPath)
    .WithEnvironment("TMP", schedulingPublisherTempPath)
    .WithEnvironment("TMPDIR", schedulingPublisherTempPath)
    .WithEnvironment("Umbraco__CMS__Hosting__LocalTempStorageLocation", "EnvironmentTemp")
    .WithEnvironment("Umbraco__CMS__Examine__LuceneDirectoryFactory", "SyncedTempFileSystemDirectoryFactory")
    .WithEnvironment("ConnectionStrings__umbracoDbDSN", umbracoDb)
    .WithEnvironment("ConnectionStrings__distributedCacheDbDSN", distributedCacheDb)
    .WithEnvironment("DistributedCache__UseSqlServerInLocal", "true")
    .WithEnvironment("ReverseProxy__UseForwardedHeaders", "true")
    .WaitFor(umbracoDb)
    .WaitFor(distributedCacheDb)
    .WithEndpoint("http", endpoint =>
    {
        endpoint.IsProxied = false;
        endpoint.TargetPort = 54201;
    })
    .WithEndpoint("https", endpoint =>
    {
        endpoint.IsProxied = false;
        endpoint.TargetPort = 64301;
    })
    .WithExternalHttpEndpoints();

var subscriber = builder.AddProject<Projects.Sst_Cms_Web_UI>("cms-web-ui-subscriber", launchProfileName: "Umbraco.Web.UI.Subscriber")
    .WithEnvironment("ASPNETCORE_ENVIRONMENT", "Local")
    .WithEnvironment("DOTNET_ENVIRONMENT", "Local")
    .WithEnvironment("UMBRACO_SERVER_ROLE", "CD")
    .WithEnvironment("TEMP", subscriberTempPath)
    .WithEnvironment("TMP", subscriberTempPath)
    .WithEnvironment("TMPDIR", subscriberTempPath)
    .WithEnvironment("Umbraco__CMS__Hosting__LocalTempStorageLocation", "EnvironmentTemp")
    .WithEnvironment("Umbraco__CMS__Examine__LuceneDirectoryFactory", "TempFileSystemDirectoryFactory")
    .WithEnvironment("ConnectionStrings__umbracoDbDSN", umbracoDb)
    .WithEnvironment("ConnectionStrings__distributedCacheDbDSN", distributedCacheDb)
    .WithEnvironment("DistributedCache__UseSqlServerInLocal", "true")
    .WithEnvironment("ReverseProxy__UseForwardedHeaders", "true")
    .WaitFor(umbracoDb)
    .WaitFor(distributedCacheDb)
    .WithEndpoint("http", endpoint =>
    {
        endpoint.IsProxied = false;
        endpoint.TargetPort = 34101;
    })
    .WithEndpoint("https", endpoint =>
    {
        endpoint.IsProxied = false;
        endpoint.TargetPort = 44101;
    })
    .WithExternalHttpEndpoints();

builder.AddProject<Projects.Sst_AppHost_ReverseProxy>("cms-dev-proxy", launchProfileName: "https")
    .WithEnvironment(context =>
    {
        context.EnvironmentVariables["ReverseProxy__Clusters__cms__Destinations__cms__Address"] = cms.GetEndpoint("http");
        context.EnvironmentVariables["ReverseProxy__Clusters__subscriber__Destinations__subscriber__Address"] = subscriber.GetEndpoint("http");
    })
    .WithReference(cms)
    .WithReference(subscriber)
    .WaitFor(cms)
    .WaitFor(subscriber)
    .WithExternalHttpEndpoints();

builder.Build().Run();

static bool IsEnabled(string environmentVariableName)
{
    var value = Environment.GetEnvironmentVariable(environmentVariableName);

    return bool.TryParse(value, out var enabled) && enabled;
}

static string GetSqlDistributedCacheCreationScript() =>
    """
    IF NOT EXISTS (SELECT 1 FROM sys.databases WHERE name = N'sst-local-distributed-cache')
    BEGIN
        CREATE DATABASE [sst-local-distributed-cache];
    END
    GO

    USE [sst-local-distributed-cache];
    GO

    IF OBJECT_ID(N'[dbo].[DistributedCache]', N'U') IS NULL
    BEGIN
        CREATE TABLE [dbo].[DistributedCache](
            [Id] nvarchar(449) COLLATE SQL_Latin1_General_CP1_CS_AS NOT NULL,
            [Value] varbinary(max) NOT NULL,
            [ExpiresAtTime] datetimeoffset NOT NULL,
            [SlidingExpirationInSeconds] bigint NULL,
            [AbsoluteExpiration] datetimeoffset NULL,
            CONSTRAINT [PK_DistributedCache] PRIMARY KEY CLUSTERED ([Id] ASC)
        );

        CREATE NONCLUSTERED INDEX [IX_DistributedCache_ExpiresAtTime]
            ON [dbo].[DistributedCache]([ExpiresAtTime] ASC);
    END
    ELSE IF NOT EXISTS (
        SELECT 1
        FROM sys.indexes
        WHERE name = N'IX_DistributedCache_ExpiresAtTime'
            AND object_id = OBJECT_ID(N'[dbo].[DistributedCache]', N'U')
    )
    BEGIN
        CREATE NONCLUSTERED INDEX [IX_DistributedCache_ExpiresAtTime]
            ON [dbo].[DistributedCache]([ExpiresAtTime] ASC);
    END
    """;
